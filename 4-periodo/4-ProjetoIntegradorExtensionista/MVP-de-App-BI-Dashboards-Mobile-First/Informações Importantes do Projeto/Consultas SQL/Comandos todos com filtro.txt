# 2º Indicador Principal — Total de Colaboradores:

# Total (para uma unidade/ano específico)

FROM rhdataset

    `Data Afastamento` = '00/00/0000'
    OR STR_TO_DATE(`Data Afastamento`, '%d/%m/%Y') >= STR_TO_DATE(CONCAT(@ano, '-01-01'), '%Y-%m-%d')
)
AND (
    @unidade = 'TODAS'
    OR `Filial` = @unidade
);

# 3º Distribuição por Gênero:

# Homens × Mulheres
-- DISTRIBUIÇÃO POR GÊNERO NO ANO
SELECT
    `Sexo`,
    COUNT(*) AS quantidade,
    ROUND(
        100 * COUNT(*) /
        (
            SELECT COUNT(*)
            FROM rhdataset
            WHERE
                STR_TO_DATE(`Admissão`, '%d/%m/%Y') <= STR_TO_DATE(CONCAT(@ano, '-12-31'), '%Y-%m-%d')
            AND (
                `Data Afastamento` = '00/00/0000'
                OR STR_TO_DATE(`Data Afastamento`, '%d/%m/%Y') >= STR_TO_DATE(CONCAT(@ano, '-01-01'), '%Y-%m-%d')
            )
            AND (
                @unidade = 'TODAS'
                OR `Filial` = @unidade
            )
        )
    ,1) AS percentual
FROM rhdataset
WHERE
    STR_TO_DATE(`Admissão`, '%d/%m/%Y') <= STR_TO_DATE(CONCAT(@ano, '-12-31'), '%Y-%m-%d')
AND (
    `Data Afastamento` = '00/00/0000'
    OR STR_TO_DATE(`Data Afastamento`, '%d/%m/%Y') >= STR_TO_DATE(CONCAT(@ano, '-01-01'), '%Y-%m-%d')
)
AND (
    @unidade = 'TODAS'
    OR `Filial` = @unidade
)
GROUP BY `Sexo`;

# 5º Colaboradores por Setor (Gráfico):

SELECT 
    `Descrição (C.Custo)` AS setor,
    COUNT(*) AS quantidade
FROM rhdataset
WHERE
    STR_TO_DATE(`Admissão`, '%d/%m/%Y') <= STR_TO_DATE(CONCAT(@ano, '-12-31'), '%Y-%m-%d')
AND (
    `Data Afastamento` = '00/00/0000'
    OR STR_TO_DATE(`Data Afastamento`, '%d/%m/%Y') >= STR_TO_DATE(CONCAT(@ano, '-01-01'), '%Y-%m-%d')
)
AND (
    @unidade = 'TODAS'
    OR `Filial` = @unidade
)
GROUP BY `Descrição (C.Custo)`
ORDER BY quantidade DESC;

# 6º Lista Resumida de Colaboradores:

SELECT
    `Nome`,
    `Descrição (C.Custo)` AS setor,
    `Título Reduzido (Cargo)` AS cargo,

    CASE
        WHEN `Descrição (Situação)` = 'Demitido'
             AND YEAR(STR_TO_DATE(`Data Afastamento`, '%d/%m/%Y')) = @ano
            THEN 'Demitido'

        WHEN `Descrição (Situação)` IN (
            'Lic. s/ Remuneração','Ferias','Lic.Medica - 15 Dias',
            'Lic.Medica - 30 Dias Prof.','Licença Maternidade',
            'Licença Paternidade','Licença Paternidade Prof.','Auxilio Doenca'
        )
        AND YEAR(STR_TO_DATE(`Data Afastamento`, '%d/%m/%Y')) = @ano
            THEN 'Com licença'
        ELSE 'Ativo'
    END AS status

FROM rhdataset
WHERE
    STR_TO_DATE(`Admissão`, '%d/%m/%Y') <= STR_TO_DATE(CONCAT(@ano, '-12-31'), '%Y-%m-%d')
AND (
    `Data Afastamento` = '00/00/0000'
    OR STR_TO_DATE(`Data Afastamento`, '%d/%m/%Y') >= STR_TO_DATE(CONCAT(@ano, '-01-01'), '%Y-%m-%d')
)
AND (
    @unidade = 'TODAS'
    OR `Filial` = @unidade
)
ORDER BY `Nome`;
